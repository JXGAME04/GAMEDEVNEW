  652| function Line(x,x1,y1,x2,y2)

  662| --当x1或者x2 < 0 ,y =0

  775| --函数GetSkillLevelData(levelname, data, level)
  776| --levelname：魔法属性名称
  777| --data：技能名称
  778| --level：技能等级
  779| --return：当技能名称为data，技能等级为level
  780| --			时的魔法属性levelname所需求的三个参数的具体值
  781| -----------------------------------------------------------
  782| function GetSkillLevelData(levelname, data, level)
  783| 	if(data==nil) then
  784| 		return ""
  785| 	end
  786| 	if(data == "") then
  787| 		return ""
  788| 	end
  789| 	if(SKILLS[data]==nil) then
  790| 		return ""
  791| 	end
  792| 	if(SKILLS[data][levelname]==nil) then
  793| 		return ""
  794| 	end
  795| 	if(type(SKILLS[data][levelname]) == "function") then
  796| 		return SKILLS[data][levelname](level)
  797| 	end
  798| 	if(SKILLS[data][levelname][1]==nil) then
  799| 		SKILLS[data][levelname][1]={{0,0},{20,0}}
  800| 	end
  801| 	if(SKILLS[data][levelname][2]==nil) then
  802| 		SKILLS[data][levelname][2]={{0,0},{20,0}}
  803| 	end
  804| 	if(SKILLS[data][levelname][3]==nil) then
  805| 		SKILLS[data][levelname][3]={{0,0},{20,0}}
  806| 	end
  807| 	p1=floor(Link(level,SKILLS[data][levelname][1]))
  808| 	p2=floor(Link(level,SKILLS[data][levelname][2]))
  809| 	p3=floor(Link(level,SKILLS[data][levelname][3]))
  810| 	return Param2String(p1,p2,p3)
  811| end;
