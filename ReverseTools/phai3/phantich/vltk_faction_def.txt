 1091|         return
 1092|     end
 1093|     if not tbFacDef.tbSkills[nFacId][nLv] then
 1094|         return
 1095|     end
 1096| 
 1097|     for _,tbSkillInfo in tbFacDef.tbSkills[nFacId][nLv] do
 1098|         local nSkillId = 0
 1099|         local nSkillLv = 0
 1100|         if type(tbSkillInfo) == "table" then
 1101|             nSkillId = tbSkillInfo[1]
 1102|             nSkillLv = tbSkillInfo[2]
 1103|         elseif type(tbSkillInfo) == "number" then
 1104|             nSkillId = tbSkillInfo
 1105|             nSkillLv = 0
 1106|         end
 1107|         if HaveMagic(nSkillId) == -1 then
 1108|             AddMagic(nSkillId,nSkillLv)
 1109|             AddSkillEvent(nFacId, nLv, nSkillId, nSkillLv)
 1110|         end
 1111|     end
