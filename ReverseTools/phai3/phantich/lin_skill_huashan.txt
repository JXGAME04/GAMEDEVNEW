  636| --根据2个点，求线形函数f(x)=k*x+b
  637| --y= (y2-y1)*(x-x1)/(x2-x1)+y1
  638| --当x2=x1, 有x=c,该直线是一条垂直于x轴的直线
  639| --这是可以取得y=任意值
  640| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  641| --函数为：
  642| function Line(x,x1,y1,x2,y2)
  643| 	if(x2==x1) then
  644| 		return y2
  645| 	end
  646| 	return (y2-y1)*(x-x1)/(x2-x1)+y1
  647| end
  648| 
  649| -----------------------------------------------
  650| --根据2个点，求2次形函数f(x)=a*x2+c
  651| --y= (y2-y1)*x*x/(x2*x2-x1*x1)-(y2-y1)*x1*x1/(x2*x2-x1*x1)+y1
  652| --当x1或者x2 < 0 ,y =0
  653| --当x2=x1, 有x=c,是一条垂直于x轴的直线
  654| --这是可以取得y=任意值
  655| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  656| --函数为：extrac
  657| function Conic(x,x1,y1,x2,y2)
  658| 	if((x1 < 0) or (x2<0))then 
  659| 		return 0
  660| 	end
  661| 	if(x2==x1) then
  662| 		return y2
  663| 	end
  664| 	return (y2-y1)*x*x/(x2*x2-x1*x1)-(y2-y1)*x1*x1/(x2*x2-x1*x1)+y1
  665| end
  666| 
  667| -----------------------------------------------
  668| --根据2个点，求-2次形函数f(x)=a*sqrt(x2)+c
  669| --y=(y2-y1)*x/(sqrt(x2)-sqrt(x1))+y1-(y2-y1)/((sqrt(x2)-sqrt(x1))
  670| --当x2或者x1<0, y=0,
  671| --当x1=x2,有x=c,是一条垂直于x轴的直线
  672| --这是可以取得y=任意值
  673| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  674| --函数为：extrac
  675| function Extrac(x,x1,y1,x2,y2)
  676| 	if((x1 < 0) or (x2<0))then 
  677| 		return 0
  678| 	end
  679| 	if(x2==x1) then
  680| 		return y2
  681| 	end

  686| --描绘连接线:Link(x,points)

  692| function Link(x,points)
  693| 	num = getn(points)
  694| 	if(num<2) then
  695| 		return -1
  696| 	end
  697| 	for i=1,num do
  698| 		if(points[i][3]==nil) then

  701| 	end

  703| 		return points[1][3](x,points[1][1],points[1][2],points[2][1],points[2][2])
  704| 	end
  705| 	if(x > points[num][1]) then
  706| 		return points[num][3](x,points[num-1][1],points[num-1][2],points[num][1],points[num][2])
  707| 	end
  708| 	
  709| 	c = 2

  713| 			break
  714| 		end
  715| 	end
  716| 	return points[c][3](x,points[c-1][1],points[c-1][2],points[c][1],points[c][2])

  765| --函数GetSkillLevelData(levelname, data, level)
  766| --levelname：魔法属性名称
  767| --data：技能名称
  768| --level：技能等级
  769| --return：当技能名称为data，技能等级为level
  770| --			时的魔法属性levelname所需求的三个参数的具体值
  771| -----------------------------------------------------------
  772| function GetSkillLevelData(levelname, data, level)
  773| 	if(data==nil) then
  774| 		return ""
  775| 	end
  776| 	if(data == "") then
  777| 		return ""
  778| 	end
  779| 	if(SKILLS[data]==nil) then
  780| 		return ""
  781| 	end
  782| 	if(SKILLS[data][levelname]==nil) then
  783| 		return ""
  784| 	end
  785| 	if(type(SKILLS[data][levelname]) == "function") then
  786| 		return SKILLS[data][levelname](level)
  787| 	end
  788| 	if(SKILLS[data][levelname][1]==nil) then
  789| 		SKILLS[data][levelname][1]={{0,0},{20,0}}
  790| 	end
  791| 	if(SKILLS[data][levelname][2]==nil) then
  792| 		SKILLS[data][levelname][2]={{0,0},{20,0}}
  793| 	end
  794| 	if(SKILLS[data][levelname][3]==nil) then
  795| 		SKILLS[data][levelname][3]={{0,0},{20,0}}
  796| 	end
  797| 	p1=floor(Link(level,SKILLS[data][levelname][1]))
  798| 	p2=floor(Link(level,SKILLS[data][levelname][2]))
  799| 	p3=floor(Link(level,SKILLS[data][levelname][3]))
  800| 	return Param2String(p1,p2,p3)
  801| end;
