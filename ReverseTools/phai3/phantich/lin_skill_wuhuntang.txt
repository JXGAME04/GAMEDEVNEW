  726| 		return ""
  727| 	end
  728| 	if(data == "") then
  729| 		return ""
  730| 	end
  731| 	if(SKILLS[data]==nil) then
  732| 		return ""
  733| 	end
  734| 	if(SKILLS[data][levelname]==nil) then
  735| 		return ""
  736| 	end
  737| 	if(type(SKILLS[data][levelname]) == "function") then
  738| 		return SKILLS[data][levelname](level)

  740| 	if(SKILLS[data][levelname][1]==nil) then
  741| 		SKILLS[data][levelname][1]={{0,0},{20,0}}

  743| 	if(SKILLS[data][levelname][2]==nil) then

  747| 		SKILLS[data][levelname][3]={{0,0},{20,0}}
  748| 	end
  749| 	p1=floor(Link(level,SKILLS[data][levelname][1]))
  750| 	p2=floor(Link(level,SKILLS[data][levelname][2]))
  751| 	p3=floor(Link(level,SKILLS[data][levelname][3]))
