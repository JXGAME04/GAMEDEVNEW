  660| --根据2个点，求线形函数f(x)=k*x+b
  661| --y= (y2-y1)*(x-x1)/(x2-x1)+y1
  662| --当x2=x1, 有x=c,该直线是一条垂直于x轴的直线
  663| --这是可以取得y=任意值
  664| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  665| --函数为：
  666| function Line(x,x1,y1,x2,y2)
  667| 	if(x2==x1) then
  668| 		return y2
  669| 	end
  670| 	return (y2-y1)*(x-x1)/(x2-x1)+y1
  671| end
  672| 
  673| -----------------------------------------------
  674| --根据2个点，求2次形函数f(x)=a*x2+c
  675| --y= (y2-y1)*x*x/(x2*x2-x1*x1)-(y2-y1)*x1*x1/(x2*x2-x1*x1)+y1
  676| --当x1或者x2 < 0 ,y =0
  677| --当x2=x1, 有x=c,是一条垂直于x轴的直线
  678| --这是可以取得y=任意值
  679| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  680| --函数为：extrac
  681| function Conic(x,x1,y1,x2,y2)
  682| 	if((x1 < 0) or (x2<0))then 
  683| 		return 0
  684| 	end
  685| 	if(x2==x1) then
  686| 		return y2
  687| 	end
  688| 	return (y2-y1)*x*x/(x2*x2-x1*x1)-(y2-y1)*x1*x1/(x2*x2-x1*x1)+y1
  689| end

  692| --根据2个点，求-2次形函数f(x)=a*sqrt(x2)+c
  693| --y=(y2-y1)*x/(sqrt(x2)-sqrt(x1))+y1-(y2-y1)/((sqrt(x2)-sqrt(x1))
  694| --当x2或者x1<0, y=0,
  695| --当x1=x2,有x=c,是一条垂直于x轴的直线
  696| --这是可以取得y=任意值
  697| --因此，如果已知两点(x1,y1),(x2,y2)可求得过此2点的
  698| --函数为：extrac
  699| function Extrac(x,x1,y1,x2,y2)
  700| 	if((x1 < 0) or (x2<0))then 
  701| 		return 0
  702| 	end
  703| 	if(x2==x1) then
  704| 		return y2
  705| 	end
  706| 	return (y2-y1)*(x-x1)/(x2-x1)+y1
  707| end
